# Backend dependency manifest for the Used Car Marketplace FastAPI service.
#
# Target runtime: Python 3.12 (CI targets 3.9; code is compatible through 3.12).
# This project uses Pydantic v1 (app/core/config.py imports `BaseSettings`
# directly from `pydantic`), so pydantic is intentionally pinned below 2.0.
#
# This manifest was introduced during environment setup; the repository
# previously shipped without one even though the CI workflow expects
# `pip install -r requirements.txt`.

# --- Web framework (Pydantic v1 era) ---
fastapi==0.95.2
uvicorn==0.23.2
python-multipart==0.0.6
httpx==0.24.1

# --- Data validation / settings management (Pydantic v1 is REQUIRED) ---
# Pydantic is kept on the 1.x line because config.py imports BaseSettings from
# `pydantic`. Version 1.10.26 is the latest 1.x patch and is required for
# compatibility with Python 3.12.4+ (the ForwardRef._evaluate recursive_guard
# change); earlier 1.10.x releases fail to import FastAPI on this interpreter.
pydantic==1.10.26
python-dotenv==1.0.1

# --- Authentication / security ---
python-jose[cryptography]==3.3.0
passlib==1.7.4
# passlib 1.7.4 cannot read the version of bcrypt >= 4.1, so bcrypt is pinned
# to 4.0.1 to keep password hashing working (per the technical specification).
bcrypt==4.0.1

# --- Google Cloud services ---
google-cloud-firestore>=2.13,<3
google-cloud-storage>=2.10,<3
google-cloud-vision>=3.4,<4
google-cloud-documentai>=2.20,<3

# --- Document / image processing ---
PyPDF2==3.0.1
Pillow==10.4.0

# --- Payments ---
stripe==7.8.0

# --- Background tasks ---
celery==5.4.0

# --- Development / CI (linting and tests) ---
pytest==8.3.3
flake8==7.1.1
